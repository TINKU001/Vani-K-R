# AI-Driven Early Cardiovascular Disease Risk Prediction
# Data Cleaning - UCI Cleveland Heart Disease Dataset
# ============================================================

import pandas as pd
import numpy as np

# ------------------------------------------------------------
# 1. Load Dataset
# ------------------------------------------------------------

# Update this path based on your file location
file_path = "/content/heart_disease_raw.csv"

columns = [
    "age",
    "sex",
    "cp",
    "trestbps",
    "chol",
    "fbs",
    "restecg",
    "thalach",
    "exang",
    "oldpeak",
    "slope",
    "ca",
    "thal",
    "num"
]

df = pd.read_csv(
    file_path,
    names=columns,
    na_values="?"
)

# Drop the first row if it's a header that was mistakenly read as data
# Check if the first row is still the header based on initial load
if df.iloc[0].tolist() == columns:
    df = df.iloc[1:].copy()

print("\n========== ORIGINAL DATASET ==========")
print("Rows:", df.shape[0])
print("Columns:", df.shape[1])
print(df.head())


# ------------------------------------------------------------
# 2. Check Data Types
# ------------------------------------------------------------

print("\n========== DATA TYPES ==========")
print(df.dtypes)


# ------------------------------------------------------------
# 3. Check Missing Values
# ------------------------------------------------------------

print("\n========== MISSING VALUES ==========")

missing_count = df.isnull().sum()
missing_percentage = (missing_count / len(df)) * 100

missing_summary = pd.DataFrame({
    "Missing Count": missing_count,
    "Missing Percentage": missing_percentage.round(2)
})

print(missing_summary)


# ------------------------------------------------------------
# 4. Check Duplicate Records
# ------------------------------------------------------------

print("\n========== DUPLICATE RECORDS ==========")

duplicate_count = df.duplicated().sum()
duplicate_percentage = (duplicate_count / len(df)) * 100

print("Duplicate Records:", duplicate_count)
print("Duplicate Percentage:", round(duplicate_percentage, 2), "%")


# ------------------------------------------------------------
# 5. Remove Duplicate Records
# ------------------------------------------------------------

df = df.drop_duplicates()

print("\nRows after removing duplicates:", len(df))


# ------------------------------------------------------------
# 6. Convert Numeric Columns
# ------------------------------------------------------------

numeric_columns = [
    "age",
    "trestbps",
    "chol",
    "thalach",
    "oldpeak",
    "ca",
    "num" # 'num' needs to be numeric for binary classification
]

for col in numeric_columns:
    df[col] = pd.to_numeric(df[col], errors="coerce")


# ------------------------------------------------------------
# 7. Check Missing Values After Conversion
# ------------------------------------------------------------

print("\n========== MISSING VALUES AFTER CONVERSION ==========")

print(df.isnull().sum())


# ------------------------------------------------------------
# 8. Handle Missing Values
# ------------------------------------------------------------

# Numerical variables - Median Imputation

for col in numeric_columns:
    if df[col].isnull().sum() > 0:
        df[col] = df[col].fillna(df[col].median())


# Categorical variables - Mode Imputation

categorical_columns = [
    "sex",
    "cp",
    "fbs",
    "restecg",
    "exang",
    "slope",
    "thal"
]

for col in categorical_columns:
    if df[col].isnull().sum() > 0:
        df[col] = df[col].fillna(df[col].mode()[0])


# ------------------------------------------------------------
# 9. Verify Missing Values
# ------------------------------------------------------------

print("\n========== MISSING VALUES AFTER IMPUTATION ==========")

print(df.isnull().sum())


# ------------------------------------------------------------
# 10. Check Target Variable Distribution
# ------------------------------------------------------------

print("\n========== ORIGINAL TARGET DISTRIBUTION ==========")

print(df["num"].value_counts().sort_index())

print("\nTarget Percentage:")

print(
    (df["num"].value_counts(normalize=True) * 100)
    .round(2)
)


# ------------------------------------------------------------
# 11. Convert Target to Binary Classification
# ------------------------------------------------------------

# 0 = No Heart Disease
# 1,2,3,4 = Heart Disease Present

df["target"] = df["num"].apply(
    lambda x: 0 if x == 0 else 1
)

# Remove original target column

df = df.drop(columns=["num"])


# ------------------------------------------------------------
# 12. Check New Target Distribution
# ------------------------------------------------------------

print("\n========== BINARY TARGET DISTRIBUTION ==========")

target_count = df["target"].value_counts().sort_index()

target_percentage = (
    df["target"].value_counts(normalize=True).sort_index() * 100
)

target_summary = pd.DataFrame({
    "Count": target_count,
    "Percentage": target_percentage.round(2)
})

print(target_summary)


# ------------------------------------------------------------
# 13. Outlier Detection Using IQR
# ------------------------------------------------------------

print("\n========== OUTLIER DETECTION ==========")

outlier_summary = []

# Iterate only through actual numeric feature columns, not the target 'num' (which is dropped)
# Remove 'num' from numeric_columns here, as it's already converted to 'target' and dropped
# from the DataFrame for outlier detection, which should only be for features.
feature_numeric_columns = [col for col in numeric_columns if col != 'num']

for col in feature_numeric_columns:

    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)

    IQR = Q3 - Q1

    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR

    outliers = df[
        (df[col] < lower_bound) |
        (df[col] > upper_bound)
    ]

    count = len(outliers)
    percentage = (count / len(df)) * 100

    outlier_summary.append([
        col,
        count,
        round(percentage, 2),
        lower_bound,
        upper_bound
    ])


outlier_df = pd.DataFrame(
    outlier_summary,
    columns=[
        "Variable",
        "Outlier Count",
        "Outlier Percentage",
        "Lower Bound",
        "Upper Bound"
    ]
)

print(outlier_df)


# ------------------------------------------------------------
# 14. Final Data Quality Check
# ------------------------------------------------------------

print("\n========== FINAL DATASET ==========")

print("Final Rows:", df.shape[0])
print("Final Columns:", df.shape[1])

print("\nRemaining Missing Values:")
print(df.isnull().sum())

print("\nDuplicate Records:")
print(df.duplicated().sum())


# ------------------------------------------------------------
# 15. Save Cleaned Dataset
# ------------------------------------------------------------

output_file = "heart_disease_cleaned.csv"

df.to_csv(
    output_file,
    index=False
)

print("\nCleaned dataset saved as:", output_file)
